cmake_minimum_required(VERSION 3.16)
project(private_coop_relay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(ROOM "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/gpu/bbnet/room")
add_library(private_room STATIC "${ROOM}/room_host.cpp")
target_include_directories(private_room PUBLIC "${ROOM}")
target_link_libraries(private_room PUBLIC nlohmann_json::nlohmann_json OpenSSL::Crypto Threads::Threads)
add_executable(private-coop-relay main.cpp)
target_link_libraries(private-coop-relay PRIVATE private_room)
if(WIN32)
    target_link_libraries(private_room PUBLIC ws2_32)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(private-room-test
        ../../engine/tests/test_room_host.cpp
        "${ROOM}/room_link.cpp" "${ROOM}/room_crypto.cpp" "${ROOM}/room_ws.cpp")
    target_include_directories(private-room-test PRIVATE ../../engine
        ../../engine/gpu/bbnet/third_party/httplib)
    target_compile_definitions(private-room-test PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
    target_link_libraries(private-room-test PRIVATE private_room OpenSSL::SSL)
    if(WIN32)
        target_link_libraries(private-room-test PRIVATE crypt32)
    endif()
    add_test(NAME private-room COMMAND private-room-test)
    set_tests_properties(private-room PROPERTIES TIMEOUT 180)
endif()
